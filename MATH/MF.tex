\section{Complexes et trigonométrie}
\subsection*{Inégalité triangulaire}
\[ ||z|-|z'|| \leq |z \pm z'| \leq |z|+|z'| \]

\subsection*{Racines troisième de l'unité}
\[j=e^{i\frac{2\pi}{3}} = -\frac{1}{2} + i \frac{\sqrt{3}}{2}\]
\[x^2 + x + 1 = (x - j)(x - j^2)\]
\[x^2 - x + 1 = (x + j)(x + j^2)\]

\subsection*{Formules trigonométriques}
\begin{align*}
    \cos{(a+b)} &= \cos{a}\cos{b}-\sin{a}\sin{b} \\
    \cos{(2a)} &= \cos^2{a} - \sin^2{a} \\
    &= 2\cos^2{a} -1 \\
    &= 1- 2\sin^2{a} \\
    \sin{(a+b)} &= \sin{a}\cos{b}+\cos{a}\sin{b} \\
    \sin{(2a)} &= 2\sin{a}\cos{a} \\
    \tan{(a+b)} &= \frac{\tan{a}+\tan{b}}{1-\tan{a}\tan{b}}
\end{align*}

\subsubsection*{Formules de linéarisation}
\begin{align*}
    \cos{a}\cos{b} &= \frac{1}{2}(\cos{(a+b)}+\cos{(a-b)}) \\
    \sin{a}\sin{b} &= \frac{1}{2}(\cos{(a-b)}-\cos{(a+b)}) \\
    \sin{a}\cos{b} &= \frac{1}{2}(\sin{(a+b)}+\sin{(a-b)})
\end{align*}

\subsubsection*{Formules de factorisation}
\begin{align*}
    \cos{p}+\cos{q} &= 2\cos{\frac{p+q}{2}}\cos{\frac{p-q}{2}} \\
    \cos{p}-\cos{q} &= -2\sin{\frac{p+q}{2}}\sin{\frac{p-q}{2}} \\
    \sin{p}+\sin{q} &= 2\sin{\frac{p+q}{2}}\cos{\frac{p-q}{2}} \\
    \sin{p}-\sin{q} &= 2\cos{\frac{p+q}{2}}\sin{\frac{p-q}{2}}
\end{align*}

\section{Sommes, produits et probabilités}
\subsection*{Sommes classiques}

\begin{align*}
    \sum_{k=0}^{n}k &= \frac{(n)(n+1)}{2} \\
    \sum_{k=0}^{n}k^2 &= \frac{(n)(n+1)(2n+1)}{6} \\
    \sum_{k=0}^{n}k^3 &= \frac{(n)^2(n+1)^2}{4} \\
    \sum_{k=n}^{p}u_k &= \frac{1}{2}(u_p+u_n)(p-n+1) \\
    \sum_{k=0}^n q^k &= 
        \begin{cases}
            \frac{1-q^{n+1}}{1-q} & \text{si } q \neq 1 \\
            n+1 & \text{sinon}
        \end{cases}
\end{align*}

\subsection*{Formule de Bernoulli}
\[a^n - b^n = (a-b)\sum_{k=0}^{n-1}(a^kb^{n-1-k}) \]

\subsection*{Formule du binôme de Newton}
\[ (a+b)^n = \sum_{k=0}^{n}\binom{n}{k} a^kb^{n-k} \]

\subsection*{Espérance}
\[ \mathbb{E}(X) = \sum_{\omega \in \Omega} X(\omega)P({\omega}) = \sum_{x \in X(\Omega)} xP(X=x) \]

\subsection*{Variance}
\[ \mathbb{V}(X) = \mathbb{E}((X-\mathbb{E}(X))^2) = \mathbb{E}(X^2) - \mathbb{E}(X)^2 \]

\subsection*{Covariance}
\[ \text{Cov}(X,Y) = \mathbb{E}((X-\mathbb{E}(X))(Y-\mathbb{E}(Y))) = \mathbb{E}(XY) - \mathbb{E}(X)\mathbb{E}(Y) \]

\subsection*{Inégalité de Markov}
Pour $a > 0$,
\[ P(|X| \geq a) \leq \frac{\mathbb{E}(|X|)}{a} \]

\subsection*{Inégalité de Bienaymé-Tchebychev}
Pour $\epsilon > 0$
\[ P( | X - \mathbb{E}(X) | \geq \epsilon ) \leq \frac{\mathbb{V}(X)}{\epsilon^2} \]

\section{Développement limité}

\subsection*{Formule de Taylor-Young}
Pour $f \in \mathcal{C}^{n}$
\[ f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!} (x-a)^k + \underset{a}{o} ((x-a)^n) \]

\subsection*{Formule de Taylor}
Pour $f \in \mathcal{C}^{n+1}$
\[ f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k + \int_{a}^{x} \frac{(x-t)^n}{n!}f^{(n+1)} (t) \dt \]

\subsection*{Inégalité de Taylor-Lagrange}
\[ \left |f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k \right | \leq \frac{|x-a|^{n+1}}{(n+1)!} M_{n+1} \]

\subsection*{Algorithme de Gram-Schmidt}
\begin{align*}
    f_1 &= \frac{e_1}{\|e_1\|} \\
    f_k &= \frac{g_k}{\|g_k\|} \\
    \text{ où } g_k &= e_k - \sum_{i=1}^{k-1} \langle e_k, f_i \rangle f_i = e_k - \sum_{i=1}^{k-1} \frac{1}{\|g_i\|^2} \langle e_k, g_i \rangle g_i
\end{align*}

\section{Polynômes}
\subsection*{Relations coefficients racines}
Pour $P = a_n X^n + a_{n-1} X^{n-1} + \dots + a_1 X + a_0$ avec les racines $r_1, r_2, \dots, r_n$ (avec multiplicité), 
\[ \sum_{i=1}^{n} r_i = -\frac{a_{n-1}}{a_n} \]
\[ \prod_{i=1}^{n} r_i = (-1)^n \frac{a_0}{a_n} \]

\subsubsection*{Cas particuliers}
\begin{itemize}
    \item Degré 2 - $x_1 + x_2 = -\frac{b}{a} \et x_1x_2 = \frac{c}{a}$

    \begin{itemize}
        \item $\sigma_1 = x_1 + x_2 + x_3$
        \item $\sigma_2 = x_1x_2 + x_1x_3 + x_2x_3$
        \item $\sigma_3 = x_1 x_2 x_3$
    \end{itemize}
\end{itemize}
