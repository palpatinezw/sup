\newcounter{quotecounter}

% List of used quotes
\def\usedquotes{}

% -----------------
% Declare a quote
% -----------------
\newcommand{\declarequote}[3]{%
  \expandafter\newcommand\csname quote@#1\endcsname{#2}%
  \expandafter\newcommand\csname source@#1\endcsname{#3}%
  \expandafter\gdef\csname refs@#1\endcsname{}%
}

\newcommand{\vml}{Vingt milles lieues sous les mers}
\newcommand{\lmi}{Le mur invisible}
\newcommand{\gcg}{La connaissance de la vie}

% -----------------
% Register quote (only once)
% -----------------
\newcommand{\registerquote}[1]{%
  \ifcsundef{used@#1}{%
    \expandafter\gdef\csname used@#1\endcsname{}%
    \ifx\usedquotes\empty
      \xdef\usedquotes{#1}%
    \else
      \xdef\usedquotes{\usedquotes,#1}%
    \fi
  }{}
}

% -----------------
% Use a quote
% -----------------
\newcommand{\usequote}[1]{%
    \stepcounter{quotecounter}%
    \hypertarget{quote:#1:\thequotecounter}{}%
        - \textit{\csname quote@#1\endcsname}
        - (\csname source@#1\endcsname)
    \registerquote{#1}%
    \expandafter\gappto\csname refs@#1\endcsname{%
        \hyperlink{quote:#1:\thequotecounter}{\thepage}, %
    }% 
}

% -----------------
% Print all references automatically (robust version)
% -----------------
\newcommand{\printallquotes}{%
  \section*{Citations}
  \begingroup
  % Split comma-separated list manually
  \def\do##1{%
    \ifstrempty{##1}{}{% skip empty
      \paragraph{}
    \noindent \csname quote@##1\endcsname
      \textit{ - \csname source@##1\endcsname}
      
      (\textit{cf} p\csname refs@##1\endcsname)
    }%
  }%
  \def\listtemp{}%
  \edef\listtemp{\usedquotes}%
  \edef\doitems{\noexpand\docsvlist{\listtemp}}%
  \doitems
  \endgroup
}
